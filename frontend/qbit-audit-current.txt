589: async function normalizeUploadedLogo(file: File): Promise<string> {
590:   const rawDataUrl = await readFileAsDataUrl(file);
591: 
592:   if (file.type === "image/svg+xml") {
593:     return rawDataUrl;
594:   }
595: 
596:   return new Promise((resolve) => {
597:     const image = new Image();
598: 
599:     image.onload = () => {
600:       const maxDimension = 1400;
601:       const largestSide = Math.max(image.width, image.height);
602:       const scale = largestSide > maxDimension ? maxDimension / largestSide : 1;
603: 
604:       const canvas = document.createElement("canvas");
605:       canvas.width = Math.max(1, Math.round(image.width * scale));
606:       canvas.height = Math.max(1, Math.round(image.height * scale));
607: 
608:       const context = canvas.getContext("2d");
609: 
610:       if (!context) {
611:         resolve(rawDataUrl);
612:         return;
613:       }
614: 
615:       context.clearRect(0, 0, canvas.width, canvas.height);
616:       context.drawImage(image, 0, 0, canvas.width, canvas.height);
617: 
618:       resolve(canvas.toDataURL("image/png"));
619:     };
620: 
621:     image.onerror = () => {
622:       resolve(rawDataUrl);
623:     };
624: 
625:     image.src = rawDataUrl;
626:   });
627: }

358: function hslToHex(hue: number, saturation: number, lightness: number) {
359:   const normalizedHue = hue / 360;
360:   const normalizedSaturation = saturation / 100;
361:   const normalizedLightness = lightness / 100;
362: 
363:   if (normalizedSaturation === 0) {
364:     const grayValue = Math.round(normalizedLightness * 255)
365:       .toString(16)
366:       .padStart(2, "0");
367: 
368:     return `#${grayValue}${grayValue}${grayValue}`;
369:   }
370: 
371:   function hueToRgb(p: number, q: number, t: number) {
372:     let adjustedT = t;
373: 
374:     if (adjustedT < 0) {
375:       adjustedT += 1;
376:     }
377: 
378:     if (adjustedT > 1) {
379:       adjustedT -= 1;
380:     }
381: 
382:     if (adjustedT < 1 / 6) {
383:       return p + (q - p) * 6 * adjustedT;
384:     }
385: 
386:     if (adjustedT < 1 / 2) {
387:       return q;
388:     }
389: 
390:     if (adjustedT < 2 / 3) {
391:       return p + (q - p) * (2 / 3 - adjustedT) * 6;
392:     }
393: 
394:     return p;
395:   }
396: 
397:   const q =
398:     normalizedLightness < 0.5
399:       ? normalizedLightness * (1 + normalizedSaturation)
400:       : normalizedLightness +
401:         normalizedSaturation -
402:         normalizedLightness * normalizedSaturation;
403: 
404:   const p = 2 * normalizedLightness - q;
405: 
406:   const red = Math.round(hueToRgb(p, q, normalizedHue + 1 / 3) * 255);
407:   const green = Math.round(hueToRgb(p, q, normalizedHue) * 255);
408:   const blue = Math.round(hueToRgb(p, q, normalizedHue - 1 / 3) * 255);
409: 
410:   return `#${red.toString(16).padStart(2, "0")}${green
411:     .toString(16)
412:     .padStart(2, "0")}${blue.toString(16).padStart(2, "0")}`;
413: }

568: function normalizeAdvertisingBlock(block: unknown): AdvertisingBlock | null {
569:   if (!block || typeof block !== "object") {
